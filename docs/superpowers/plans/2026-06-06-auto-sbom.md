# Auto-SBOM Orchestration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `packmon scan --auto-sbom` so the CLI detects a project's manifests, runs the matching CycloneDX generator(s), and feeds the generated SBOM(s) into the existing scan pipeline in one command.

**Architecture:** A new isolated package `internal/sbomgen` (Detector + per-ecosystem Generator adapters + Orchestrator) produces temporary CycloneDX JSON files. `cmd/packmon` appends their paths to `scanSettings.SBOMFiles`, which already flows into `scanner.Config.SBOMFiles` → `scanner.CollectPackages`. Nothing downstream of that append changes. The orchestrator exposes injectable `LookPath`/`Runner` seams so all logic is unit-testable without the real tools.

**Tech Stack:** Go, `os/exec` (no shell), `spf13/cobra`, `golang.org/x/mod/modfile`, `encoding/xml`, `github.com/BurntSushi/toml`, existing `internal/sbom` (`sbom.Parse`, `sbom.IsCycloneDXJSON`), `internal/scanner` (`CollectPackages`).

**Design source:** `docs/superpowers/specs/2026-06-06-auto-sbom-design.md`.

package sbomgen

import (
	"encoding/json"
	"encoding/xml"
	"io/fs"
	"os"
	"path"
	"path/filepath"
	"strings"

	"github.com/BurntSushi/toml"
)

// Detection is one concrete SBOM generation target.
type Detection struct {
	Ecosystem    string // "go", "npm", "pypi", "maven"
	ProjectDir   string // absolute dir where the generator runs
	ManifestPath string // absolute path to the manifest that triggered detection
	InputKind    string // "go.mod", "npm-package", "requirements", "poetry", "maven-pom"
	DisplayPath  string // manifest path relative to the scan root, for messages
}

var skipDirs = map[string]struct{}{
	".git": {}, "node_modules": {}, "vendor": {}, "__pycache__": {},
}

// Detect walks root (bounded by maxDepth) and returns generation targets.
func Detect(root string, maxDepth int) ([]Detection, error) {
	absRoot, err := filepath.Abs(root)
	if err != nil {
		return nil, err
	}

	var manifests []string // absolute paths
	err = filepath.WalkDir(absRoot, func(p string, d fs.DirEntry, err error) error {
		if err != nil {
			return nil // skip unreadable entries
		}
		if d.IsDir() {
			name := d.Name()
			if p != absRoot {
				if _, skip := skipDirs[name]; skip || strings.HasPrefix(name, ".") {
					return filepath.SkipDir
				}
				if depthExceeded(absRoot, p, maxDepth) {
					return filepath.SkipDir
				}
			}
			return nil
		}
		switch d.Name() {
		case "go.mod", "package.json", "requirements.txt", "pyproject.toml",
			"Pipfile", "pom.xml":
			manifests = append(manifests, p)
		}
		return nil
	})
	if err != nil {
		return nil, err
	}

	var detections []Detection
	for _, m := range manifests {
		det, ok := classifyManifest(absRoot, m)
		if ok {
			detections = append(detections, det)
		}
	}
	return suppressCovered(absRoot, detections), nil
}

func depthExceeded(root, dir string, maxDepth int) bool {
	rel, err := filepath.Rel(root, dir)
	if err != nil {
		return false
	}
	if rel == "." {
		return false
	}
	return strings.Count(rel, string(filepath.Separator))+1 > maxDepth
}

func classifyManifest(root, manifest string) (Detection, bool) {
	dir := filepath.Dir(manifest)
	display := relDisplay(root, manifest)
	base := filepath.Base(manifest)
	switch base {
	case "go.mod":
		return Detection{"go", dir, manifest, "go.mod", display}, true
	case "package.json":
		return Detection{"npm", dir, manifest, "npm-package", display}, true
	case "requirements.txt":
		return Detection{"pypi", dir, manifest, "requirements", display}, true
	case "pom.xml":
		return Detection{"maven", dir, manifest, "maven-pom", display}, true
	case "pyproject.toml":
		if isPoetryProject(manifest) {
			return Detection{"pypi", dir, manifest, "poetry", display}, true
		}
		return Detection{}, false // bare pyproject: deferred (PDM/uv)
	default:
		return Detection{}, false // Pipfile: deferred in v1
	}
}

func relDisplay(root, p string) string {
	rel, err := filepath.Rel(root, p)
	if err != nil {
		return p
	}
	return rel
}

func isPoetryProject(pyproject string) bool {
	data, err := os.ReadFile(pyproject) // #nosec G304 -- path comes from a controlled walk
	if err != nil {
		return false
	}
	var doc struct {
		Tool struct {
			Poetry struct {
				Name         string         `toml:"name"`
				Dependencies map[string]any `toml:"dependencies"`
			} `toml:"poetry"`
		} `toml:"tool"`
	}
	if err := toml.Unmarshal(data, &doc); err != nil {
		return false
	}
	return strings.TrimSpace(doc.Tool.Poetry.Name) != "" || len(doc.Tool.Poetry.Dependencies) > 0
}

// suppressCovered removes child detections covered by an npm workspace root or a
// Maven aggregator root. Suppression is computed from manifest contents and is
// scoped by ecosystem AND directory, so an npm workspace only suppresses npm
// child manifests -- a co-located go.mod/pom.xml in the same directory survives.
func suppressCovered(root string, ds []Detection) []Detection {
	suppressed := map[string]struct{}{}
	key := func(ecosystem, dir string) string { return ecosystem + "\x00" + dir }

	for _, d := range ds {
		switch d.Ecosystem {
		case "npm":
			for _, child := range npmWorkspaceChildren(d) {
				suppressed[key("npm", child)] = struct{}{}
			}
		case "maven":
			for _, child := range mavenModuleChildren(d) {
				suppressed[key("maven", child)] = struct{}{}
			}
		}
	}

	out := make([]Detection, 0, len(ds))
	for _, d := range ds {
		if _, covered := suppressed[key(d.Ecosystem, d.ProjectDir)]; covered {
			continue
		}
		out = append(out, d)
	}
	return out
}

func npmWorkspaceChildren(d Detection) []string {
	data, err := os.ReadFile(d.ManifestPath) // #nosec G304 -- controlled walk
	if err != nil {
		return nil
	}
	var pkg struct {
		Workspaces json.RawMessage `json:"workspaces"`
	}
	if json.Unmarshal(data, &pkg) != nil || len(pkg.Workspaces) == 0 {
		return nil
	}
	var globs []string
	if json.Unmarshal(pkg.Workspaces, &globs) != nil {
		var obj struct {
			Packages []string `json:"packages"`
		}
		if json.Unmarshal(pkg.Workspaces, &obj) != nil {
			return nil
		}
		globs = obj.Packages
	}

	var children []string
	for _, g := range globs {
		// Match a child project dir (relative to the workspace root) against the glob.
		base := filepath.ToSlash(d.ProjectDir)
		full := path.Join(base, g)
		matches, _ := filepath.Glob(filepath.FromSlash(full))
		for _, m := range matches {
			info, statErr := os.Stat(m)
			if statErr == nil && info.IsDir() {
				children = append(children, m)
			}
		}
	}
	return children
}

func mavenModuleChildren(d Detection) []string {
	return mavenModulesRecursive(d.ProjectDir)
}

func mavenModulesRecursive(dir string) []string {
	return mavenModulesWalk(dir, map[string]struct{}{})
}

// mavenModulesWalk follows <modules> recursively, guarding against cyclic or
// self-referential module declarations via a visited set.
func mavenModulesWalk(dir string, visited map[string]struct{}) []string {
	abs, err := filepath.Abs(dir)
	if err != nil {
		abs = dir
	}
	if _, seen := visited[abs]; seen {
		return nil
	}
	visited[abs] = struct{}{}

	data, err := os.ReadFile(filepath.Join(dir, "pom.xml")) // #nosec G304 -- controlled walk
	if err != nil {
		return nil
	}
	var project struct {
		Modules []string `xml:"modules>module"`
	}
	if err := xml.Unmarshal(data, &project); err != nil {
		return nil
	}
	var children []string
	for _, mod := range project.Modules {
		mod = strings.TrimSpace(mod)
		if mod == "" {
			continue
		}
		childDir := filepath.Join(dir, filepath.FromSlash(mod))
		children = append(children, childDir)
		children = append(children, mavenModulesWalk(childDir, visited)...)
	}
	return children
}

package sbomgen

import "context"

// Pinned generator versions. Revise only together with the spec table and tests.
const (
	npmGeneratorVersion   = "4.2.1"
	pypiGeneratorVersion  = "7.3.0"
	goGeneratorVersion    = "v1.10.0"
	mavenPluginVersion    = "2.9.1"
)

// RunOptions describes one external command invocation. Dir is optional; when
// set, the command runs from that working directory.
type RunOptions struct {
	Name string
	Args []string
	Dir  string
}

// RunnerFunc runs an external command and returns combined output. Injected so
// tests never spawn real processes.
type RunnerFunc func(ctx context.Context, opts RunOptions) ([]byte, error)

// GenerateOptions controls a single SBOM generation.
type GenerateOptions struct {
	IncludeDev bool
}

// InstallSpec describes how to obtain a missing generator.
type InstallSpec struct {
	Package        string   // human-readable package id
	Source         string   // registry/source for disclosure
	Args           []string // full argv (no shell); empty when CanAutoInstall is false
	CanAutoInstall bool
}

// Generator builds a CycloneDX SBOM for one ecosystem.
type Generator interface {
	Ecosystem() string
	Tool() string
	InstallSpec() InstallSpec
	Generate(ctx context.Context, d Detection, outPath string, opts GenerateOptions, run RunnerFunc) error
	// DeclaresDependencies reports whether the detection's manifest lists any
	// dependencies, for the zero-package cross-check.
	DeclaresDependencies(d Detection) (bool, error)
}

// DefaultRegistry maps ecosystem name to its generator.
func DefaultRegistry() map[string]Generator {
	return map[string]Generator{
		"go":    goGenerator{},
		"npm":   npmGenerator{},
		"pypi":  pypiGenerator{},
		"maven": mavenGenerator{},
	}
}

package sbomgen

import (
	"bytes"
	"fmt"
	"os"

	"github.com/8linkz/packmon/internal/sbom"
)

// validateGeneratedSBOM runs the structural checks from the spec and returns the
// number of importable packages plus skipped-component diagnostics. A zero
// package count is NOT an error here; the caller applies the cross-check.
func validateGeneratedSBOM(path string) (int, []sbom.SkippedComponent, error) {
	data, err := os.ReadFile(path) // #nosec G304 -- path is generator output we just created
	if err != nil {
		return 0, nil, fmt.Errorf("read generated SBOM: %w", err)
	}
	if len(bytes.TrimSpace(data)) == 0 {
		return 0, nil, fmt.Errorf("generated SBOM is empty")
	}
	if !sbom.IsCycloneDXJSON(data) {
		return 0, nil, fmt.Errorf("generated file is not CycloneDX JSON")
	}
	parsed, err := sbom.Parse(bytes.NewReader(data))
	if err != nil {
		return 0, nil, fmt.Errorf("parse generated SBOM: %w", err)
	}
	return len(parsed.Packages), parsed.Skipped, nil

package sbomgen

import (
	"context"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"fmt"
	"log/slog"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"time"

// Config drives a single auto-SBOM run.
type Config struct {
	Target       string               // single resolved scan target directory
	Ecosystems   []string             // lowercase filter; empty = all
	InstallTools bool                 // allow auto-install of missing generators
	KeepSBOMDir  string               // "" = temp mode; else keep here
	IncludeDev   bool                 // honored at generation time
	MaxDepth     int                  // detector walk depth (default 10 if 0)
	Timeout      time.Duration        // per-generator (default 2m if 0)
	Logger       *slog.Logger         // nil -> discard
	Registry     map[string]Generator // nil -> DefaultRegistry()
	LookPath     func(string) (string, error)
	Runner       RunnerFunc
}

// Result is the orchestrator output.
type Result struct {
	SBOMPaths []string
	Cleanup   func() error
}

func defaultLookPath(name string) (string, error) { return exec.LookPath(name) }

func defaultRunner(ctx context.Context, opts RunOptions) ([]byte, error) {
	cmd := exec.CommandContext(ctx, opts.Name, opts.Args...) // #nosec G204 -- name/args are built from fixed adapters and pinned specs, never user shell input
	cmd.Dir = opts.Dir
	cmd.WaitDelay = 2 * time.Second
	return cmd.CombinedOutput()
}

func (c *Config) normalize() {
	if c.Registry == nil {
		c.Registry = DefaultRegistry()
	}
	if c.LookPath == nil {
		c.LookPath = defaultLookPath
	}
	if c.Runner == nil {
		c.Runner = defaultRunner
	}
	if c.Logger == nil {
		c.Logger = slog.New(slog.NewTextHandler(os.Stderr, &slog.HandlerOptions{Level: slog.LevelError}))
	}
	if c.MaxDepth <= 0 {
		c.MaxDepth = 10
	}
	if c.Timeout <= 0 {
		c.Timeout = 2 * time.Minute
	}
}

// Run detects, generates, validates, and returns SBOM paths with a cleanup
// callback. On any error, generated files are cleaned up before returning.
func Run(ctx context.Context, cfg Config) (Result, error) {
	cfg.normalize()

	detections, err := Detect(cfg.Target, cfg.MaxDepth)
	if err != nil {
		return Result{}, fmt.Errorf("detect: %w", err)
	}
	detections = filterEcosystems(detections, cfg.Ecosystems)
	if len(detections) == 0 {
		return Result{}, fmt.Errorf("no supported manifest found under %s (looked for: go.mod, package.json, requirements.txt, Poetry pyproject.toml, pom.xml)", cfg.Target)
	}

	outDir, cleanup, err := prepareOutputDir(cfg.KeepSBOMDir)
	if err != nil {
		return Result{}, err
	}

	var paths []string
	var created []string // every outPath touched, for keep-mode failure cleanup
	fail := func(e error) (Result, error) {
		// Temp mode: cleanup nukes the whole dir. Keep mode: cleanup is a no-op,
		// so remove any files we created this run (including reserved-but-failed
		// ones) — a hard-failed run must not leave partial SBOMs behind.
		if cfg.KeepSBOMDir != "" {
			for _, p := range created {
				_ = os.Remove(p)
			}
		}
		_ = cleanup()
		return Result{}, e
	}

	for _, d := range detections {
		gen, ok := cfg.Registry[d.Ecosystem]
		if !ok {
			return fail(fmt.Errorf("%s: no generator registered", d.Ecosystem))
		}
		if err := ensureTool(ctx, cfg, gen, d); err != nil {
			return fail(err)
		}

		outPath := filepath.Join(outDir, sbomFileName(cfg.Target, d))
		if cfg.KeepSBOMDir != "" {
			if err := reserveOutputPath(outPath); err != nil {
				return fail(fmt.Errorf("%s: %w", d.DisplayPath, err))
			}
			created = append(created, outPath)
		}

		genCtx, cancel := context.WithTimeout(ctx, cfg.Timeout)
		genErr := gen.Generate(genCtx, d, outPath, GenerateOptions{IncludeDev: cfg.IncludeDev}, cfg.Runner)
		cancel()
		if genErr != nil {
			return fail(fmt.Errorf("%s: generate failed: %w", d.DisplayPath, genErr))
		}

		count, skipped, valErr := validateGeneratedSBOM(outPath)
		if valErr != nil {
			return fail(fmt.Errorf("%s: %w", d.DisplayPath, valErr))
		}
		for _, s := range skipped {
			cfg.Logger.Warn("skipped SBOM component", slog.String("detection", d.DisplayPath), slog.String("name", s.Name), slog.String("reason", s.Reason))
		}
		if count == 0 {
			declares, dErr := gen.DeclaresDependencies(d)
			if dErr != nil {
				return fail(fmt.Errorf("%s: dependency cross-check: %w", d.DisplayPath, dErr))
			}
			if declares {
				return fail(fmt.Errorf("%s: generator produced zero packages but the manifest declares dependencies (likely a soft failure, e.g. a missing install step)", d.DisplayPath))
			}
			cfg.Logger.Warn("detection contributed zero packages (dependency-free project)", slog.String("detection", d.DisplayPath))
		}

		paths = append(paths, outPath)
	}

	return Result{SBOMPaths: paths, Cleanup: cleanup}, nil
}

func filterEcosystems(ds []Detection, filter []string) []Detection {
	if len(filter) == 0 {
		return ds
	}
	allow := map[string]struct{}{}
	for _, e := range filter {
		allow[strings.ToLower(strings.TrimSpace(e))] = struct{}{}
	}
	out := ds[:0]
	for _, d := range ds {
		if _, ok := allow[d.Ecosystem]; ok {
			out = append(out, d)
		}
	}
	return out
}

func prepareOutputDir(keepDir string) (dir string, cleanup func() error, err error) {
	if keepDir != "" {
		if err := os.MkdirAll(keepDir, 0o700); err != nil {
			return "", nil, fmt.Errorf("create --keep-sbom dir: %w", err)
		}
		return keepDir, func() error { return nil }, nil // no-op: kept files persist
	}
	tmp, err := os.MkdirTemp("", "packmon-sbom-")
	if err != nil {
		return "", nil, fmt.Errorf("create temp dir: %w", err)
	}
	return tmp, func() error { return os.RemoveAll(tmp) }, nil
}

func reserveOutputPath(path string) error {
	f, err := os.OpenFile(path, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o600)
	if err != nil {
		if errors.Is(err, os.ErrExist) {
			return fmt.Errorf("refusing to overwrite existing %s", path)
		}
		return fmt.Errorf("reserve output %s: %w", path, err)
	}
	return f.Close()
}

func sbomFileName(root string, d Detection) string {
	rel, err := filepath.Rel(root, d.ProjectDir)
	if err != nil || rel == "." {
		rel = "root"
	}
	safe := strings.NewReplacer(string(filepath.Separator), "-", "/", "-", " ", "_").Replace(rel)
	sum := sha256.Sum256([]byte(d.ManifestPath))
	short := hex.EncodeToString(sum[:])[:8]
	return fmt.Sprintf("sbom-%s-%s-%s.cdx.json", d.Ecosystem, safe, short)
}

func ensureTool(ctx context.Context, cfg Config, gen Generator, d Detection) error {
	if _, err := cfg.LookPath(gen.Tool()); err == nil {
		return nil
	} else if !errors.Is(err, exec.ErrNotFound) && !errors.Is(err, os.ErrNotExist) {
		// Some platforms return other errors; treat as missing but keep the message.
		cfg.Logger.Debug("LookPath error treated as missing", slog.String("tool", gen.Tool()), slog.String("error", err.Error()))
	}

	spec := gen.InstallSpec()
	if !spec.CanAutoInstall {
		return fmt.Errorf("%s: %q not found. %s must be installed manually (Packmon cannot auto-install it)", d.Ecosystem, gen.Tool(), gen.Tool())
	}
	if !cfg.InstallTools {
		return fmt.Errorf("%s: generator %q not found in PATH.\n    Install manually:  %s\n    Or automatically:  add --install-tools",
			d.Ecosystem, gen.Tool(), strings.Join(spec.Args, " "))
	}
	if len(spec.Args) == 0 {
		return fmt.Errorf("%s: install spec for %q has no argv", d.Ecosystem, spec.Package)
	}
	installer := spec.Args[0]
	if _, err := cfg.LookPath(installer); err != nil {
		return fmt.Errorf("%s: installer %q not found; install %s manually from %s", d.Ecosystem, installer, spec.Package, spec.Source)
	}

	cfg.Logger.Warn("installing generator",
		slog.String("ecosystem", d.Ecosystem),
		slog.String("package", spec.Package),
		slog.String("source", spec.Source),
		slog.String("command", strings.Join(spec.Args, " ")),
	)
	installCtx, cancel := context.WithTimeout(ctx, cfg.Timeout)
	defer cancel()
	if out, err := cfg.Runner(installCtx, RunOptions{Name: spec.Args[0], Args: spec.Args[1:]}); err != nil {
		return fmt.Errorf("%s: install %q failed: %w\n%s", d.Ecosystem, spec.Package, err, string(out))
	}
	if _, err := cfg.LookPath(gen.Tool()); err != nil {
		return fmt.Errorf("%s: %q still not found after installing %s; ensure the install location is on PATH", d.Ecosystem, gen.Tool(), spec.Package)
	}
	return nil
}

func (pypiGenerator) Generate(ctx context.Context, d Detection, outPath string, _ GenerateOptions, run RunnerFunc) error {
	var args []string
	dir := ""
	switch d.InputKind {
	case "poetry":
		args = []string{"poetry", "--output-format", "JSON", "--output-file", outPath}
		dir = d.ProjectDir
	default: // "requirements"
		args = []string{"requirements", "--output-format", "JSON", "--output-file", outPath, d.ManifestPath}
	}
	if out, err := run(ctx, RunOptions{Name: "cyclonedx-py", Args: args, Dir: dir}); err != nil {
		return fmt.Errorf("cyclonedx-py: %w\n%s", err, string(out))
	}

func (pypiGenerator) DeclaresDependencies(d Detection) (bool, error) {
	data, err := os.ReadFile(d.ManifestPath) // #nosec G304 -- controlled detection path
	if err != nil {
		return false, err
	}
	if d.InputKind == "poetry" {
		var doc struct {
			Tool struct {
				Poetry struct {
					Dependencies map[string]any `toml:"dependencies"`
					Group        map[string]struct {
						Dependencies map[string]any `toml:"dependencies"`
					} `toml:"group"`
				} `toml:"poetry"`
			} `toml:"tool"`
		}
		if _, err := toml.Decode(string(data), &doc); err != nil {
			return false, err
		}
		if hasPoetryDependencies(doc.Tool.Poetry.Dependencies) {
			return true, nil
		}
		for _, group := range doc.Tool.Poetry.Group {
			if hasPoetryDependencies(group.Dependencies) {
				return true, nil
			}
		}
		if poetryLockDeclaresDependencies(d.ProjectDir) {
			return true, nil
		}
		return false, nil
	}
	for _, line := range strings.Split(string(data), "\n") {
		line, _, _ = strings.Cut(line, "#")
		line = strings.TrimSpace(line)
		if line == "" {
			continue
		}
		if isRequirementsOptionOnlyLine(strings.Fields(line)[0]) {
			continue
		}
		return true, nil
	}
	return false, nil
}

func hasPoetryDependencies(deps map[string]any) bool {
	for name := range deps {
		if strings.ToLower(strings.TrimSpace(name)) != "python" {
			return true
		}
	}
	return false
}

func poetryLockDeclaresDependencies(projectDir string) bool {
	data, err := os.ReadFile(filepath.Join(projectDir, "poetry.lock")) // #nosec G304 -- controlled detection path
	return err == nil && strings.TrimSpace(string(data)) != ""
}

func isRequirementsOptionOnlyLine(first string) bool {
	if first == "--editable" || strings.HasPrefix(first, "--editable=") {
		return false
	}
	switch first {
	case "-i", "--index-url", "--extra-index-url", "-f", "--find-links", "--trusted-host",
		"--require-hashes", "--pre", "--prefer-binary", "-c", "--constraint", "-r", "--requirement":
		return true
	}
	for _, prefix := range []string{
		"--index-url=", "--extra-index-url=", "--find-links=", "--trusted-host=", "--constraint=", "--requirement=",
	} {
		if strings.HasPrefix(first, prefix) {
			return true
		}
	}
	return strings.HasPrefix(first, "--")
}

	return InstallSpec{Package: "cyclonedx-maven-plugin", Source: "Maven Central", CanAutoInstall: false}
}

func (mavenGenerator) Generate(ctx context.Context, d Detection, outPath string, opts GenerateOptions, run RunnerFunc) error {
	stageDir, err := os.MkdirTemp("", "packmon-mvn-")
	if err != nil {
		return fmt.Errorf("maven stage dir: %w", err)
	}
	defer func() { _ = os.RemoveAll(stageDir) }()

	const name = "bom"
	args := []string{
		"-q", "-f", filepath.Join(d.ProjectDir, "pom.xml"),
		"org.cyclonedx:cyclonedx-maven-plugin:" + mavenPluginVersion + ":makeAggregateBom",
		"-DoutputFormat=json",
		"-DoutputDirectory=" + stageDir,
		"-DoutputName=" + name,
	}
	if opts.IncludeDev {
		args = append(args, "-DincludeTestScope=true")
	}
	if out, err := run(ctx, RunOptions{Name: "mvn", Args: args}); err != nil {
		return fmt.Errorf("mvn: %w\n%s", err, string(out))
	}

	staged := filepath.Join(stageDir, name+".json")
	data, err := os.ReadFile(staged) // #nosec G304 -- staged path we constructed
	if err != nil {
		return fmt.Errorf("maven plugin did not produce %s: %w", name+".json", err)
	}
	if err := os.WriteFile(outPath, data, 0o600); err != nil {
		return fmt.Errorf("copy maven bom: %w", err)
	}
	return nil
}

func (mavenGenerator) DeclaresDependencies(d Detection) (bool, error) {
	if has, err := mavenPOMDeclaresDirectDependencies(d.ManifestPath); err != nil || has {
		return has, err
	}
	// Aggregator root with no direct deps: check covered module poms.
	for _, child := range mavenModulesRecursive(d.ProjectDir) {
		has, err := mavenPOMDeclaresDirectDependencies(filepath.Join(child, "pom.xml"))
		if err != nil {

		}
	}

}

	data, err := os.ReadFile(pomPath) // #nosec G304 -- controlled detection path

func validateAutoSBOMOnlySettings(settings scanSettings) error {
	if len(settings.SBOMFiles) > 0 {
		return fmt.Errorf("--sbom-only cannot be combined with --sbom because no scan is run")
	}
	if strings.TrimSpace(settings.OutputJSON) != "" ||
		strings.TrimSpace(settings.OutputSARIF) != "" ||
		strings.TrimSpace(settings.OutputJUnit) != "" ||
		strings.TrimSpace(settings.OutputHTML) != "" {
		return fmt.Errorf("--sbom-only cannot be combined with report outputs because no scan result is produced")
	}
	if strings.TrimSpace(settings.WebhookURL) != "" || strings.TrimSpace(settings.WebhookSecret) != "" {
		return fmt.Errorf("--sbom-only cannot be combined with webhook output because no scan result is produced")
	}
	return nil
}

func buildAutoSBOMConfig(settings scanSettings, auto autoSBOMFlags) sbomgen.Config {
	keepDir := strings.TrimSpace(auto.KeepSBOM)
	if auto.SBOMOnly && keepDir == "" {
		keepDir = "."
	}
	return sbomgen.Config{
		Target:       settings.Path,
		Ecosystems:   append([]string(nil), settings.Ecosystems...),
		InstallTools: auto.InstallTools,
		KeepSBOMDir:  keepDir,
		IncludeDev:   settings.IncludeDev,
		MaxDepth:     settings.MaxDepth,
		Logger:       scanLogger(settings.Quiet, settings.LogLevel),
	}
}

func runAutoSBOMCommand(cmd *cobra.Command, args []string, flags scanFlagValues, auto autoSBOMFlags) error {
	return runAutoSBOMCommandWithDeps(cmd, args, flags, auto, defaultAutoSBOMDeps)
}

func runAutoSBOMCommandWithDeps(cmd *cobra.Command, args []string, flags scanFlagValues, auto autoSBOMFlags, deps autoSBOMDeps) error {
	cfg, _, err := deps.loadConfig()
	if err != nil {
		return withExitCode(ExitOperational, err)
	}
	targets, err := buildScanTargets(cfg, args, flags)
	if err != nil {
		return withExitCode(ExitOperational, err)
	}
	if len(targets) != 1 {
		return withExitCode(ExitOperational, fmt.Errorf("--auto-sbom needs exactly one target, got %d", len(targets)))
	}
	settings, err := resolveScanSettings(cmd, cfg, targets[0], flags)
	if err != nil {
		return withExitCode(ExitOperational, err)
	}

	if auto.SBOMOnly {
		if err := validateAutoSBOMOnlySettings(settings); err != nil {
			return withExitCode(ExitOperational, err)
		}
	}

	res, err := deps.generate(cmd.Context(), buildAutoSBOMConfig(settings, auto))
	if err != nil {
		return withExitCode(ExitOperational, err)
	}
	defer func() { _ = res.Cleanup() }()

	if auto.SBOMOnly {
		if !settings.Quiet {
			for _, p := range res.SBOMPaths {
				fmt.Fprintln(os.Stdout, p)
			}
		}
		return nil
	}

	settings.SBOMFiles = append(append([]string(nil), settings.SBOMFiles...), res.SBOMPaths...)

	exitCode, err := deps.scan(cmd.Context(), settings)
	if err != nil {
		return withExitCode(exitCode, err)
	}
	if exitCode != ExitOK {
		os.Exit(exitCode)
	}
	return nil
}
